# Great question — and your instinct is exactly right.

**Honest current state:** v0.9's inclusion logic is a *static denylist* (`node_modules`, `dist`, …) + a size cap. That's **not intelligent** — it doesn't read `.gitignore`, it doesn't detect binaries/minified/generated files, and it treats a `README.md` the same as `engine.ts`. So let's fix it properly with a real **admission layer**.

## The design: `gatekeeper` — a tiered admission pipeline

The core insight: **include/exclude is the wrong question.** The right question is *"how much intelligence should we spend on this file?"* — because docs, config, and code all have value, but need different treatment.

```
FILE ARRIVES
   │
   ▼
① GIT IS TRUTH ────── `git ls-files` = the project's own answer to
   │                   "what matters". Best signal, zero config.
   │  (no git? → parse .gitignore files as fallback)
   ▼
② HARD DENY ────────── .git, .cip, binary (null-byte sniff)
   ▼
③ TRUST HEURISTICS ─── lockfiles, *.min.js, @generated / DO-NOT-EDIT
   │                    headers, minified detection (avg line > 200 chars)
   ▼
④ TIERING (not binary!)
   ├─ code     → full: symbols + edges + chunks + embeddings
   ├─ doc      → chunked by headings + embeddings, NO symbols
   ├─ config   → one chunk, embedded (scripts, docker, CI)
   └─ pathonly → file exists in the index, nothing embedded
   ▼
⑤ EXPLICIT OVERRIDES ── config.toml include/exclude ALWAYS win
```

**Why docs should be indexed (your doc question):** docs are frequently the *best* answer to "how does X work / why is it this way". But extracting "symbols" from prose is nonsense — so docs get heading-based chunking (the doc equivalent of symbols) and are tagged so search results say `"doc"` vs `"code"`.

**The agent-workflow nuance:** `git ls-files` misses *brand-new untracked files* — exactly what an agent creates mid-session. So the rule is: **tracked files ∪ (untracked but clearly source)**.

```python
"""Intelligent file admission: WHAT to index and HOW MUCH intelligence to spend.
Truth priority: git ls-files > .gitignore parse > heuristics > config overrides."""
import os, re, subprocess
from .base import repo_root, load_config, DEFAULT_EXCLUDES

HARD_DIRS = DEFAULT_EXCLUDES
DOC_EXTS = {".md", ".mdx", ".rst", ".txt", ".adoc"}
CONFIG_EXTS = {".json", ".yaml", ".yml", ".toml", ".ini", ".dockerfile", ".properties"}
CONFIG_NAMES = {"dockerfile", "makefile", ".env", "justfile", "procfile", ".editorconfig"}
ASSET_EXTS = {".png", ".jpg", ".jpeg", ".gif", ".svg", ".webp", ".ico", ".woff", ".woff2",
              ".ttf", ".eot", ".mp3", ".mp4", ".mov", ".pdf", ".zip", ".gz", ".tar",
              ".wasm", ".bin", ".exe", ".dll", ".pyc", ".sqlite", ".db"}
LOCKFILES = {"package-lock.json", "yarn.lock", "pnpm-lock.yaml", "bun.lockb", "composer.lock",
             "cargo.lock", "poetry.lock", "gemfile.lock", "go.sum", "uv.lock", "flake.lock"}
GENERATED_RES = (re.compile(r"@generated"), re.compile(r"DO NOT EDIT", re.I),
                 re.compile(r"generated by", re.I), re.compile(r"auto-?generated", re.I))

# ---------- git as truth ----------

def git_tracked(root):
    """Gold standard: exactly the files git considers worth keeping."""
    try:
        out = subprocess.run(["git", "ls-files", "-z"], cwd=root, capture_output=True, timeout=60)
        if out.returncode == 0:
            return {p for p in out.stdout.decode("utf-8", "replace").split("\0") if p}
    except Exception:
        pass
    return None

def parse_gitignore(root):
    """Fallback: merge every .gitignore (nested included) into a matcher."""
    patterns = []
    for dirpath, dirnames, _ in os.walk(root):
        dirnames[:] = [d for d in dirnames if d not in HARD_DIRS]
        if ".gitignore" in dirnames:  # never descend
            dirnames.remove(".gitignore")
        gi = os.path.join(dirpath, ".gitignore")
        if os.path.exists(gi):
            base = os.path.relpath(dirpath, root).replace(os.sep, "/")
            for line in open(gi, encoding="utf-8", errors="replace"):
                line = line.split("#", 1)[0].strip()
                if line and not line.startswith("!"):
                    patterns.append((base, line))
    return patterns

def _ignored(rel, patterns):
    for base, pat in patterns:
        p = pat.rstrip("/")
        full = p if base == "." else f"{base}/{p}"
        if (rel == full or rel.startswith(full + "/") or rel.endswith("/" + p)
                or ("/" + p + "/") in ("/" + rel + "/")):
            return True
    return False

# ---------- content heuristics ----------

def is_binary(ap):
    try:
        with open(ap, "rb") as f:
            return b"\x00" in f.read(8192)
    except OSError:
        return True

def is_minified(ap, size):
    if size > 2_000_000:
        return True
    try:
        with open(ap, encoding="utf-8", errors="replace") as f:
            head = f.read(4096)
        lines = head.splitlines() or [""]
        return len(head) > 2000 and (len(head) / len(lines)) > 200
    except OSError:
        return False

def is_generated(ap, rel):
    name = os.path.basename(rel).lower()
    if ".generated." in name or name.endswith((".min.js", ".min.css", ".bundle.js")):
        return True
    try:
        with open(ap, encoding="utf-8", errors="replace") as f:
            head = f.read(512)
        return any(rx.search(head) for rx in GENERATED_RES)
    except OSError:
        return False

def _looks_new_source(rel, ap):
    """Untracked-but-valid source: agents create files before committing them."""
    from .detect import lang_for
    if not lang_for(rel):
        return False
    try:
        with open(ap, encoding="utf-8", errors="replace") as f:
            head = f.read(512)
        return not any(rx.search(head) for rx in GENERATED_RES)
    except OSError:
        return False

# ---------- classification ----------

def classify(rel):
    """tier, reason — HOW MUCH intelligence to spend."""
    ext = os.path.splitext(rel)[1].lower()
    base = os.path.basename(rel).lower()
    if ext in ASSET_EXTS or base in LOCKFILES:
        return "skip", "asset/lockfile"
    if ext in DOC_EXTS or base == "readme":
        return "doc", "documentation"
    if ext in CONFIG_EXTS or base in CONFIG_NAMES:
        return "config", "config/infra"
    from .detect import lang_for
    if lang_for(rel):
        return "code", "source"
    return "pathonly", "unknown type"

def _decide(rel, ap, size, cfg, tracked, ign):
    include = cfg["index"].get("include", [])
    exclude = cfg["index"].get("exclude", [])
    maxb = int(cfg["index"]["max_file_kb"]) * 1024
    if any(m in rel for m in include):
        tier, _ = classify(rel)
        return "index", tier, "explicit include"
    if any(m in rel for m in exclude):
        return "skip", "-", "config exclude"
    if tracked is not None:
        if rel not in tracked and not _looks_new_source(rel, ap):
            return "skip", "-", "not tracked by git"
    elif ign is not None and _ignored(rel, ign):
        return "skip", "-", "gitignore"
    if is_binary(ap):
        return "skip", "-", "binary"
    tier, why = classify(rel)
    if tier == "skip":
        return "skip", "-", why
    if size > maxb:
        return "index", "pathonly", f"oversize ({size // 1024}KB)"
    if tier == "code" and is_generated(ap, rel):
        return "index", "pathonly", "generated code"
    if tier == "code" and is_minified(ap, size):
        return "index", "pathonly", "minified"
    return "index", tier, why

# ---------- public API ----------

def _scan(root):
    for dirpath, dirnames, filenames in os.walk(root):
        dirnames[:] = [d for d in dirnames if d not in HARD_DIRS]
        rel_dir = os.path.relpath(dirpath, root).replace(os.sep, "/")
        for fn in filenames:
            yield (fn if rel_dir == "." else f"{rel_dir}/{fn}", os.path.join(dirpath, fn))

def iter_files_smart(root, cfg):
    """The intelligent front door. Yields (rel_path, tier, why) for indexable files."""
    tracked = git_tracked(root)
    ign = None if tracked is not None else parse_gitignore(root)
    for rel, ap in _scan(root):
        try:
            size = os.path.getsize(ap)
        except OSError:
            continue
        decision, tier, why = _decide(rel, ap, size, cfg, tracked, ign)
        if decision == "index":
            yield rel, tier, why

def admission_report(root=None):
    """Full audit of every decision — the trust/transparency feature."""
    root = root or repo_root(); cfg = load_config(root)
    tracked = git_tracked(root)
    ign = None if tracked is not None else parse_gitignore(root)
    tiers, skips = {}, {}
    examples = {}
    for rel, ap in _scan(root):
        try:
            size = os.path.getsize(ap)
        except OSError:
            continue
        d, tier, why = _decide(rel, ap, size, cfg, tracked, ign)
        key = tier if d == "index" else f"skip:{why}"
        if d == "index":
            tiers[key] = tiers.get(key, 0) + 1
            examples.setdefault(key, []).append(rel)
        else:
            skips[why] = skips.get(why, 0) + 1
    return {"mode": "git-tracked" if tracked is not None else "gitignore-fallback",
            "index_tiers": tiers, "skipped": skips,
            "examples": {k: v[:3] for k, v in examples.items()}}

def explain(root, rel):
    """Why was THIS file included/excluded? `cip admission --path src/x.ts`"""
    root = root or repo_root(); cfg = load_config(root)
    ap = os.path.join(root, rel)
    if not os.path.exists(ap):
        return f"{rel}: does not exist"
    try:
        size = os.path.getsize(ap)
    except OSError:
        return f"{rel}: unreadable"
    tracked = git_tracked(root)
    ign = None if tracked is not None else parse_gitignore(root)
    d, tier, why = _decide(rel, ap, size, cfg, tracked, ign)
    return f"{rel}: {d.upper()} as '{tier}' — {why}"

# ---------- doc chunking: headings are the 'symbols' of prose ----------

def chunk_markdown(path, source, cap=50):
    chunks, cur, start, title = [], [], 1, path
    lines = source.splitlines()
    def flush(end):
        text = "\n".join(cur).strip()
        if text:
            chunks.append({"id": f"{path}#L{start}-L{end}", "path": path,
                           "symbol_id": None, "start": start, "end": end,
                           "text": text[:8000]})
    for i, line in enumerate(lines):
        if re.match(r"^#{1,3} ", line) and cur:
            flush(i)
            cur, start, title = [], i + 1, line.lstrip("# ").strip()
        cur.append(line)
    flush(len(lines))
    return chunks[:cap]
```

```python
def index_file(con, path, source, h, size, mtime, tier="code"):
    language = lang_for(path)
    parsed = None
    if tier == "code":
        parsed = parse_file(path, language, source)
    con.execute("DELETE FROM symbols WHERE path=?", (path,))
    con.execute("DELETE FROM chunks WHERE path=?", (path,))
    con.execute("DELETE FROM edges WHERE src_path=?", (path,))
    con.execute("DELETE FROM file_imports WHERE path=?", (path,))
    con.execute("DELETE FROM vectors WHERE id LIKE ?", (path + "#%",))
    con.execute("INSERT OR REPLACE INTO files"
                "(path,language,size,lines,hash,mtime,indexed_at,tier) VALUES(?,?,?,?,?,?,?,?)",
                (path, language, size, source.count("\n") + 1, h, mtime, time.time(), tier))
    if parsed:                                   # code: full treatment
        for s in parsed["symbols"]:
            con.execute("INSERT OR REPLACE INTO symbols"
                        "(id,name,kind,path,start_line,end_line,signature,body_hash,body) "
                        "VALUES(?,?,?,?,?,?,?,?,?)",
                        (s["id"], s["name"], s["kind"], path, s["start"], s["end"],
                         s["signature"], s["body_hash"], s["body"]))
            con.execute("INSERT OR IGNORE INTO edges(src,dst,kind,src_path) VALUES(?,?,?,?)",
                        (path, s["id"], "contains", path))
            if s["exported"]:
                con.execute("INSERT OR IGNORE INTO edges(src,dst,kind,src_path) VALUES(?,?,?,?)",
                            (path, s["id"], "exports", path))
        chunks = parsed["chunks"]
        for spec in parsed["imports"]:
            con.execute("INSERT INTO file_imports(path,spec) VALUES(?,?)", (path, spec))
    elif tier == "doc":                          # docs: heading-based chunks, no symbols
        chunks = chunk_markdown(path, source)
    elif tier == "config":                       # config: one capped chunk
        text = "\n".join(source.splitlines()[:60])
        from .base import sha as _sha
        chunks = [{"id": f"{path}#L1-L{min(60, max(1, source.count(chr(10)) + 1))}",
                   "path": path, "symbol_id": None, "start": 1, "end": 60, "text": text}]
    else:                                        # pathonly: metadata only, no embedding
        chunks = []
    for c in chunks:
        con.execute("INSERT OR REPLACE INTO chunks"
                    "(id,path,symbol_id,start_line,end_line,text,text_hash) VALUES(?,?,?,?,?,?,?)",
                    (c["id"], path, c.get("symbol_id"), c["start"], c["end"],
                     c["text"], _sha(c["text"])))
```

## 4. Decision-rule summary

| Rule | Decision | Why |
|---|---|---|
| `git ls-files` membership | primary gate | the repo's own curated manifest |
| untracked but valid source ext | **include** | agent-created files pre-commit |
| no git → `.gitignore` (all nested) | fallback gate | best available signal |
| null-byte sniff | skip | binaries poison embeddings |
| lockfiles, assets | skip | size, zero retrieval value |
| `@generated` / `DO NOT EDIT` / minified | pathonly | exists but no intelligence spent |
| `*.md/*.rst/docs` | **doc tier** | heading-chunked + embedded, tagged |
| config/docker/CI | config tier | one chunk — answers "what scripts/deps exist" |
| over size cap | pathonly | metadata only |

```python
            size = os.path.getsize(ap)
        except OSError:
            continue
        decision, tier, why = _decide(rel, ap, size, cfg, tracked, ign)
        if decision == "index":
            yield rel, tier, why

def admission_report(root=None):
    """Full audit of every decision — the trust/transparency feature."""
    root = root or repo_root(); cfg = load_config(root)
    tracked = git_tracked(root)
    ign = None if tracked is not None else parse_gitignore(root)
    tiers, skips = {}, {}
    examples = {}
    for rel, ap in _scan(root):
        try:
            size = os.path.getsize(ap)
        except OSError:
            continue
        d, tier, why = _decide(rel, ap, size, cfg, tracked, ign)
        key = tier if d == "index" else f"skip:{why}"
        if d == "index":
            tiers[key] = tiers.get(key, 0) + 1
            examples.setdefault(key, []).append(rel)
        else:
            skips[why] = skips.get(why, 0) + 1
    return {"mode": "git-tracked" if tracked is not None else "gitignore-fallback",
            "index_tiers": tiers, "skipped": skips,
            "examples": {k: v[:3] for k, v in examples.items()}}

def explain(root, rel):
    """Why was THIS file included/excluded? `cip admission --path src/x.ts`"""
    root = root or repo_root(); cfg = load_config(root)
    ap = os.path.join(root, rel)
    if not os.path.exists(ap):
        return f"{rel}: does not exist"
    try:
        size = os.path.getsize(ap)
    except OSError:
        return f"{rel}: unreadable"
    tracked = git_tracked(root)
    ign = None if tracked is not None else parse_gitignore(root)
    d, tier, why = _decide(rel, ap, size, cfg, tracked, ign)
    return f"{rel}: {d.upper()} as '{tier}' — {why}"

# ---------- doc chunking: headings are the 'symbols' of prose ----------

def chunk_markdown(path, source, cap=50):
    chunks, cur, start, title = [], [], 1, path
    lines = source.splitlines()
    def flush(end):
        text = "\n".join(cur).strip()
        if text:
            chunks.append({"id": f"{path}#L{start}-L{end}", "path": path,
                           "symbol_id": None, "start": start, "end": end,
                           "text": text[:8000]})
    for i, line in enumerate(lines):
        if re.match(r"^#{1,3} ", line) and cur:
            flush(i)
            cur, start, title = [], i + 1, line.lstrip("# ").strip()
        cur.append(line)
    flush(len(lines))
    return chunks[:cap]
```

## 4. Decision-rule summary

| Rule | Decision | Why |
|---|---|---|
| `git ls-files` membership | primary gate | the repo's own curated manifest |
| untracked but valid source ext | **include** | agent-created files pre-commit |
| no git → `.gitignore` (all nested) | fallback gate | best available signal |
| null-byte sniff | skip | binaries poison embeddings |
| lockfiles, assets | skip | size, zero retrieval value |
| `@generated` / `DO NOT EDIT` / minified | pathonly | exists but no intelligence spent |
| `*.md/*.rst/docs` | **doc tier** | heading-chunked + embedded, tagged |
| config/docker/CI | config tier | one chunk — answers "what scripts/deps exist" |
| over size cap | pathonly | metadata only |
| `config.toml` include/exclude | **always wins** | explicit human override |
